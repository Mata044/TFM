The following code provides an algorithm to compute the torsion of a certain family of Jacobian varieties.
\begin{verbatim}

x = QQ['x'].gen()
maxp = 41
T = prime_range(5, maxp+1)

@cached_function
def npoints(aa,bb):
	x = aa.parent()['x'].gen()
	try:
		return HyperellipticCurve(x*(x^2 - aa)*(x^2 - bb)).frobenius_polynomial()(1)
	except ValueError:
		return 0

# Check which classes survive

for a in range(2, maxp):
	for b in range(1, a):
		g = gcd([npoints(GF(p)(a)**2, GF(p)(b)**2) for p in T])
		if g != 16:
			print(a, b, g)
\end{verbatim}
